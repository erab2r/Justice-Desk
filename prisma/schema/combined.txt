model Appointment {
id              String            @id @default(uuid())

status          AppointmentStatus @default(PENDING)

joiningTime     DateTime?
serialNumber    Int?

meetingLink     String?

recordUrl       String?
recordPublicId  String?

createdAt       DateTime          @default(now())
updatedAt       DateTime          @updatedAt

// Client relation
clientId        String
client          Client            @relation(fields: [clientId],references: [id],onDelete: Cascade,onUpdate: Cascade)

// Lawyer relation
lawyerId        String
lawyer          Lawyer            @relation(fields: [lawyerId],references: [id],onDelete: Cascade,onUpdate: Cascade)

// Schedule relation
scheduleId      String
schedule        Schedule          @relation(fields: [scheduleId],references: [id],onDelete: Cascade,onUpdate: Cascade)

// Payment relation
payment         Payment?

// Case relation
case            Case?

@@unique([clientId, lawyerId, scheduleId],name: "unique_appointment")

@@unique([scheduleId, serialNumber, joiningTime],name: "unique_appointment_serial_number")

@@index([clientId], name: "idx_appointment_client")
@@index([lawyerId], name: "idx_appointment_lawyer")
@@index([scheduleId], name: "idx_appointment_schedule")
@@index([status], name: "idx_appointment_status")

@@map("appointments")
}
